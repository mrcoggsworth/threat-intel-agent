# Oracle WebLogic / CVE-2026-21962
index=* (sourcetype=access_* OR sourcetype=weblogic*) (uri_path="*/console/*" OR uri_path="*/weblogic*") (http_method=GET OR http_method=POST OR http_method=PUT OR http_method=DELETE)
| stats count values(status) as status values(uri_path) as uri_path by src_ip dest_ip host user
| where count >= 5

# miniOrange SAML / CVE-2026-15981 / CVE-2026-61979
index=* (uri_path="*mo_saml*" OR uri_path="*SAMLResponse*" OR uri_path="*/sso*")
| stats count values(user) as users values(status) as status by src_ip host uri_path
| where count >= 1

# Zimbra / CVE-2026-73570
index=* (sourcetype=zimbra* OR process_name IN (snmpd,zimbra,shell,bash,sh)) ("zimbra-snmp" OR "/opt/zimbra" OR "SNMP")
| stats count values(process_name) as process values(command_line) as command by host src_ip
