# Gitea CVE-2026-60004
index=* (url="*/diffpatch*" OR request_uri="*/diffpatch*") ("git hook" OR "hooks/" OR "git apply" OR "sh -c")
| stats count min(_time) as first_seen max(_time) as last_seen values(src_ip) as src_ip values(user) as user by host

# NemoClaw CVE-2026-65105

# Mirage2FA / Microsoft 365 session-theft hunting
index=* (OperationName="*SignIn*" OR operation="*SignIn*") ResultType=0 (AuthenticationRequirement="*multiFactorAuthentication*" OR AuthenticationRequirement="*singleFactorAuthentication*")
| stats count dc(IPAddress) as source_ip_count values(IPAddress) as source_ips values(UserAgent) as user_agents by UserPrincipalName SessionId
| where source_ip_count > 1
